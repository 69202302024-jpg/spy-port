// HEIC Profile Image Converter and Loader with Commit Hash Support async function loadProfileImage() { const imgEl = document.getElementById('profileImg'); // Candidate URLs starting with the specific commit SHA (218edf4c223dc21f8e64f3effdc7abf5535ee28b) const commitHash = '218edf4c223dc21f8e64f3effdc7abf5535ee28b'; const candidateUrls = [ `https://raw.githubusercontent.com/69202302024-jpg/spy-port/${commitHash}/IMG_3717.heic`, `https://raw.githubusercontent.com/69202302024-jpg/spy-port/${commitHash}/IMG_3717.jpg`, `https://raw.githubusercontent.com/69202302024-jpg/spy-port/${commitHash}/IMG_3717.jpeg`, `https://raw.githubusercontent.com/69202302024-jpg/spy-port/${commitHash}/IMG_3717.png`, 'https://raw.githubusercontent.com/69202302024-jpg/spy-port/main/IMG_3717.heic' ]; for (const url of candidateUrls) { try { const response = await fetch(url); if (!response.ok) continue; const blob = await response.blob(); // Convert HEIC to JPEG using heic2any for browsers that don't support HEIC natively if (window.heic2any && (blob.type.includes('heic') || url.toLowerCase().endsWith('.heic'))) { const convertedBlob = await heic2any({ blob: blob, toType: "image/jpeg", quality: 0.85 }); const resultBlob = Array.isArray(convertedBlob) ? convertedBlob[0] : convertedBlob; imgEl.src = URL.createObjectURL(resultBlob); } else { imgEl.src = URL.createObjectURL(blob); } imgEl.classList.remove('opacity-0'); return; // Stop on first successful image load } catch (err) { console.warn("Retrying next profile image URL candidate:", url, err); } } } function handleImgError(img) { loadProfileImage(); } window.addEventListener('DOMContentLoaded', loadProfileImage);
